async function backup(page: Page): Promise<SaveCurrent> {
  await navTo(page, 'settings');
  const pending = page.waitForEvent('download');
  await page.getByTestId('export-save-button').click();
  const download = await pending;
  try { return JSON.parse(await readFile(await download.path(), 'utf8')) as SaveCurrent; }
  finally { await download.delete(); }
}
async function stored(page: Page): Promise<SaveCurrent> {
  return page.evaluate(() => new Promise<SaveCurrent>((resolve, reject) => {
    const open = indexedDB.open('D2H5TextGame');
    open.onerror = () => { reject(new Error('Could not open saved character')); };
    open.onsuccess = () => {
      const db = open.result;
      const get = db.transaction('saves', 'readonly').objectStore('saves').get('current');
      get.onerror = () => { db.close(); reject(new Error('Could not read saved character')); };
      get.onsuccess = () => {
        const record = get.result as { data: SaveCurrent } | undefined;
        db.close();
        if (!record) reject(new Error('No committed character'));
        else resolve(record.data);
      };
    };
  }));
}

async function countMonsters(cards: Locator) {
  let count = 0;
  for (const card of await cards.all()) {
    const badge = card.locator(':scope > div:first-of-type > div').filter({ hasText: /^\+\d+$/ });
    expect(await badge.count()).toBeLessThanOrEqual(1);
    count += 1 + (await badge.count() ? Number((await badge.innerText()).slice(1)) : 0);
  }
  return count;
}

  const artifact=`${label}${occurrence?`-${occurrence}`:''}.json.gz`;

}

/** Failed-wave diagnostics only. Preserve the first terminal assertion even if export/comparison fails. */
async function expectTerminalOrStop(page,result,expectedWave,lastEarned,area,label,info) {
  try { expectTerminal(result,expectedWave); }
  catch(nonwin) {
    const diagnostic={primaryError:String(nonwin),area,wave:expectedWave,terminal:result,
      exportCaptured:false,zeroFailedWaveAwards:false,comparisonError:null,diagnosticErrors:[]};
    // Terminal DOM was captured synchronously before this function; screenshot also precedes navigation.
    try {await attach(info,`${label}-nonwin-terminal`,result);} catch(error) {diagnostic.diagnosticErrors.push(String(error));}
    try {await info.attach(`${label}-nonwin-screen`,{body:await page.screenshot(),contentType:'image/png'});} catch(error) {diagnostic.diagnosticErrors.push(String(error));}
    try {
      const expected=structuredClone(lastEarned);
      expected.map.currentAct=Number(/^a([12])-/.exec(area)?.[1]);
      expected.map.currentSubAreaId=area; // Ordinary Enter uses the short alias; no other expected mutation.
      const failed=await backup(page);diagnostic.exportCaptured=true;
      await attach(info,`${label}-nonwin-export`,failed);
      await attach(info,`${label}-nonwin-last-earned-expectation`,expected);
      const equal=isDeepStrictEqual;
      diagnostic.awards={
        heroXP:heroXP(failed)-heroXP(lastEarned),mercXP:mercXP(failed)-mercXP(lastEarned),
        skillPoints:failed.player.skillPoints-lastEarned.player.skillPoints,
        statPoints:failed.player.statPoints-lastEarned.player.statPoints,
        itemsUnchanged:equal(failed.inventory.backpack,lastEarned.inventory.backpack)&&equal(failed.inventory.stash,lastEarned.inventory.stash)&&equal(failed.inventory.equipped,lastEarned.inventory.equipped)&&equal(failed.mercs.mercEquipment,lastEarned.mercs.mercEquipment),
        currenciesUnchanged:equal(failed.inventory.currencies,lastEarned.inventory.currencies),
        questItemsUnchanged:equal(failed.inventory.questItems,lastEarned.inventory.questItems),
        questsUnchanged:equal(failed.map.questProgress,lastEarned.map.questProgress),
        clearsUnchanged:equal(failed.map.clearedSubAreas,lastEarned.map.clearedSubAreas),
        wholeSaveUnchangedExceptEnteredLocation:equal(failed,expected)
      };
      diagnostic.zeroFailedWaveAwards=diagnostic.awards.heroXP===0&&diagnostic.awards.mercXP===0&&
        diagnostic.awards.skillPoints===0&&diagnostic.awards.statPoints===0&&
        diagnostic.awards.itemsUnchanged&&diagnostic.awards.currenciesUnchanged&&diagnostic.awards.questItemsUnchanged&&
        diagnostic.awards.questsUnchanged&&diagnostic.awards.clearsUnchanged&&diagnostic.awards.wholeSaveUnchangedExceptEnteredLocation;
      try {expect(failed).toEqual(expected);} catch(error) {diagnostic.comparisonError=String(error);}
    } catch(error) {diagnostic.diagnosticErrors.push(String(error));}
    try {await attach(info,`${label}-nonwin-zero-awards-classification`,diagnostic);} catch(error) {diagnostic.diagnosticErrors.push(String(error));}
    throw nonwin; // Never proceed, including when diagnostic export/equality itself fails.
  }
}

    await writeFile(`${raw}/outcome.json`,JSON.stringify({outcome:'pass',classification:'fresh current-runtime earned campaign',c1FinalOrdinal:ordinal,launchCount,wallMilliseconds:Date.now()-wallStarted,level:campaign.player.level,heroXP:heroXP(campaign),mercXP:mercXP(campaign),items:allIds(campaign).length,actThreeAvailable:true,actThreeLaunched:false,errors},null,2));

  } catch(error) {
    await writeFile(`${raw}/outcome.json`,JSON.stringify({outcome:'failed',error:String(error),stack:error instanceof Error?error.stack:null,errors},null,2));
    throw error;
  } finally {
    for (const capture of [
      async () => { await attach(info,'browser-errors',errors); },
      async () => { if(!active.isClosed()) await attach(info,'last-native-DOM',await terminalDOM(active)); },
      async () => { if(!active.isClosed()) await info.attach('last-native-screen',{body:await active.screenshot(),contentType:'image/png'}); },
      async () => { if(!active.isClosed()) await attach(info,'last-native-save',await backup(active)); }
    ]) {try {await capture();} catch(error) {diagnosticFailure??=error instanceof Error?error:new Error(String(error));}}
    await helper.close();
  }
  if(diagnosticFailure) throw diagnosticFailure;
});

function heroXP(save) {
  let sum=save.player.experience;
  for(let level=1;level<save.player.level;level++) sum+=Math.floor(100*level**2.5);
  return sum;
}
function mercXP(save) {
  const actor=save.mercs.ownedMercs.find(row=>row.id===save.mercs.fieldedMercId);
  let sum=save.mercs.mercProgress[actor.id].experience;
  for(let level=1;level<actor.level;level++) sum+=Math.floor(60*level*1.3**(level-1));
  return sum;
}
function allIds(save) {
  return [...save.inventory.backpack,...save.inventory.stash,...Object.values(save.inventory.equipped).filter(Boolean),
    ...Object.values(save.mercs.mercEquipment).flatMap(row=>Object.values(row).filter(Boolean))].map(row=>row.id);
}
function xpPlayer(player, amount) {
  const state={...player};let xp=heroXP({player})+amount;let level=1;
  while(xp>=Math.floor(100*level**2.5)) {xp-=Math.floor(100*level**2.5);level++;}
  state.level=level;state.experience=xp;state.experienceToNextLevel=Math.floor(100*level**2.5);
  state.statPoints+=5*(level-player.level);
  state.skillPoints+=level-player.level+[12,24,36].filter(n=>player.level<n&&level>=n).length;
  return state;
}

function expectMercAdvance(after,before,amount) {
  const expected=structuredClone(before.mercs),id=expected.fieldedMercId;
  const actor=expected.ownedMercs.find(row=>row.id===id),progress=expected.mercProgress[id];
  const start=actor.level;let xp=progress.experience+amount;
  const threshold=level=>Math.floor(60*level*1.3**(level-1));
  while(xp>=threshold(actor.level)) {xp-=threshold(actor.level);actor.level++;}
  const gains=actor.level-start;
  // Construct only the expected actor; the browser's mercenary is never patched.
  const original=before.mercs.ownedMercs.find(row=>row.id===id);
  expected.ownedMercs[expected.ownedMercs.findIndex(row=>row.id===id)]=expectedRogueGrowth(original,gains);
  progress.experience=xp;progress.experienceToNextLevel=threshold(actor.level);
  expect(after.mercs).toEqual(expected);
  return expected;
}

// Independent expected-value arithmetic for this fixed Rogue recipe. No growth
// implementation is imported, and no returned object is written into the game.
const mercGrowthKeys=['attack','defense','attackSpeed','manaMax','physDodge'] as const;
function rogueBareFields(core,level) {
  const effective=value=>value<=200?value:200+(Math.min(500,value)-200)*.5;
  const strength=effective(core.strength),dexterity=effective(core.dexterity),energy=effective(core.energy);
  return {
    attack:Math.floor(10+(level-1)*2+Math.max(strength,dexterity)*.5),
    defense:Math.floor(20+(level-1)*3),attackSpeed:Math.max(1,Math.floor(100+dexterity)),
    manaMax:Math.floor(50+(level-1)*4+energy*1.5),physDodge:.05+dexterity*.002
  };
}
function expectedRogueGrowth(actor,levels) {
  expect(actor.id).toBe(merc);expect(actor.classId).toBe('rogue');
  expect(Number.isInteger(levels)&&levels>=0).toBe(true);
  if(levels===0) return structuredClone(actor);
  const result=structuredClone(actor),before=rogueBareFields(actor.coreStats,actor.level);
  result.level+=levels;
  result.coreStats.strength+=levels;result.coreStats.dexterity+=3*levels;result.coreStats.energy+=levels;
  const after=rogueBareFields(result.coreStats,result.level);
  for(const key of mercGrowthKeys) {
    const delta=after[key]-before[key];
    result.derivedStats[key]=delta===0?actor.derivedStats[key]:actor.derivedStats[key]+delta;
  }
  result.derivedStats.lifeMax+=8*levels;
  result.derivedStats.life=Math.min(result.derivedStats.lifeMax,actor.derivedStats.life+8*levels);
  // Explicit MR and saved current mana remain exactly as copied, without gear.
  expect(result.derivedStats.mana).toBe(actor.derivedStats.mana);
  expect(result.derivedStats.magicResist).toBe(actor.derivedStats.magicResist);
  return result;
}
async function readLootSummary(page) {
  const summary=page.getByTestId('loot-summary');await expect(summary).toBeVisible();
  return summary.evaluate(node=> {
    const clean=value=>(value??'').replace(/\s+/g,' ').trim();
    const resources=Array.from(node.querySelectorAll(':scope > div > span'),span=>span.textContent).join(' ');
    const amount=key=>Number(new RegExp('\\+(\\d+) '+key+'(?:\\s|$)').exec(resources)?.[1]??0);
    const runeShards=Number(/^\+(\d+)/.exec(node.querySelector(':scope > div')?.textContent??'')?.[1]??0);
    return {runeShards,runes:amount('rune'),gems:amount('gem'),wishstones:amount('wishstone'),
      items:Array.from(node.querySelectorAll('[data-testid="loot-items"] > li'),li=>clean(li.textContent)),text:clean(node.textContent)};
  });
}
