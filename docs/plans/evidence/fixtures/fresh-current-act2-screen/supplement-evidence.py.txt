from pathlib import Path
import base64, gzip, hashlib, json, re, subprocess

r = Path(__file__).resolve().parent
w = Path('/home/felixzhao/.zai/profiles/anonymous/worktrees/D2H5TextGame-human-worktree-2')
def read(name):
    return json.loads(gzip.decompress((r / name).read_bytes()))
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def ids(s):
    inv = s['inventory']
    items = inv['backpack'] + inv['stash'] + [x for x in inv['equipped'].values() if x]
    items += [x for g in s['mercs']['mercEquipment'].values() for x in g.values() if x]
    return [x['id'] for x in items]

original = json.loads((r / 'analysis.json').read_text())
entries = [json.loads(x) for x in (r / 'progress.jsonl').read_text().splitlines()]
report = json.loads((r / 'browser-report.json').read_text())
attachments = list(original['retainedAttachments'])
results = []
def walk(suites):
    for suite in suites:
        for spec in suite.get('specs', []):
            for test in spec.get('tests', []):
                for result in test.get('results', []):
                    results.append({k: result.get(k) for k in ['status', 'retry', 'duration']})
                    for a in result.get('attachments', []):
                        if 'body' not in a:
                            continue
                        assert a['contentType'] == 'image/png'
                        name = re.sub(r'[^a-zA-Z0-9_-]', '-', a['name']) + '.png'
                        target = r / 'retained-images' / name
                        data = base64.b64decode(a['body'], validate=True)
                        assert data.startswith(b'\x89PNG\r\n\x1a\n')
                        if target.exists():
                            assert target.read_bytes() == data
                        else:
                            target.write_bytes(data)
                        attachments.append({'name': a['name'], 'path': str(target),
                            'sha256': sha(target), 'contentType': a['contentType']})
        walk(suite.get('suites', []))
walk(report['suites'])
