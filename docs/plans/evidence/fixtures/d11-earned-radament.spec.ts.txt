async function backup(page:Page) {
 await navTo(page,'settings'); const pending=page.waitForEvent('download');
 await page.getByTestId('export-save-button').click(); const download=await pending;
 try { return JSON.parse(await readFile(await download.path(),'utf8')); } finally { await download.delete(); }
}
async function disk(page:Page) {
 return page.evaluate(async()=>{
  const db=await new Promise<IDBDatabase>((resolve,reject)=>{const req=indexedDB.open('D2H5TextGame'); req.onsuccess=()=>resolve(req.result); req.onerror=()=>reject(req.error);});
  try {
   const record=await new Promise<{data:unknown}>((resolve,reject)=>{const req=db.transaction('saves','readonly').objectStore('saves').get('current'); req.onsuccess=()=>resolve(req.result); req.onerror=()=>reject(req.error);});
   return JSON.parse(JSON.stringify(record.data,(_key,value)=>value instanceof Map?[...value.entries()]:value));
  } finally { db.close(); }
 });
}
async function importSave(page:Page, raw:string, english:boolean) {

}
async function reopen(page:Page, context:BrowserContext, english:boolean, expected:unknown, errors:string[]) {

async function save(label:string,value:unknown) {await writeFile(root+'/'+label+'.json.gz',gzipSync(JSON.stringify(value)));}
function ids(s) {return [...s.inventory.backpack,...s.inventory.stash,...Object.values(s.inventory.equipped),...Object.values(s.mercs.mercEquipment).flatMap(Object.values)].filter(Boolean).map(x=>x.id);}
function heroXP(s) {let xp=s.player.experience;for(let l=1;l<s.player.level;l++)xp+=Math.floor(100*l**2.5);return xp;}
function mercXP(s) {const actor=s.mercs.ownedMercs.find(x=>x.id===s.mercs.fieldedMercId);let xp=s.mercs.mercProgress[actor.id].experience;for(let l=1;l<actor.level;l++)xp+=Math.floor(60*l*1.3**(l-1));return xp;}
function hashSeed(text:string) {let x=0;for(let i=0;i<text.length;i++)x=((x<<5)-x+text.charCodeAt(i))>>>0;return x;}
async function wave(page:Page) {return Number((await page.getByTestId('wave-counter').innerText()).match(/\d+/)?.[0]);}
async function pause(page:Page) {await page.getByRole('button',{name:'Pause',exact:true}).click({timeout:1000});await expect(page.getByRole('button',{name:'Resume',exact:true})).toHaveAttribute('aria-pressed','true');}
async function terminal(page:Page) {return page.evaluate(()=>({
 counter:document.querySelector('[data-testid="wave-counter"]')?.textContent,
 log:Array.from(document.querySelectorAll('[data-testid="combat-log"] > div > span:last-child'),n=>n.textContent),
 enemies:Array.from(document.querySelectorAll('[data-testid="enemies-grid"] [data-testid^="unit-card-"]'),c=>({id:c.getAttribute('data-testid'),dead:c.getAttribute('data-dead'),text:c.textContent})),
 hero:document.querySelector('[data-testid="unit-card-player-1792497600000"]')?.getAttribute('data-dead'),
 defeats:document.querySelectorAll('[data-testid="defeat-panel"]').length,
 skipDisabled:document.querySelector<HTMLButtonElement>('[data-testid="skip-button"]')?.disabled
 }));}
function conservation(after,before) {
 expect(new Set(ids(after)).size).toBe(ids(after).length);const afterIds=new Set(ids(after));expect(ids(before).every(id=>afterIds.has(id))).toBe(true);
 expect(after.inventory.backpack.slice(0,before.inventory.backpack.length)).toEqual(before.inventory.backpack);
 expect(after.inventory.stash).toEqual(before.inventory.stash);expect(after.inventory.equipped).toEqual(before.inventory.equipped);
 expect(after.mercs.mercEquipment).toEqual(before.mercs.mercEquipment);expect(after.mercs.fieldedMercId).toBe(before.mercs.fieldedMercId);
}
