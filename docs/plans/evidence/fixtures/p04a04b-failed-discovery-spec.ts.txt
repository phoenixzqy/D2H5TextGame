/** Native earned Den clear → Akara → explicit claim → production save/reopen. */
import { expect, test, type Locator, type Page, type TestInfo } from '@playwright/test';
import { readFile } from 'node:fs/promises';
import type { SaveCurrent } from '../../src/engine/types/save';
import { createCharacter, navTo } from '../e2e/_helpers';

const fixedTime = new Date('2026-10-20T12:00:00.000Z');
const name = 'Earned Den';
const fire = 'skills-sorceress-fire-bolt';
const frost = 'skills-sorceress-frost-nova';
const ice = 'skills-sorceress-ice-bolt';
const merc = 'mercs/act1-rogue-inner-sight';
const quest = 'quests/act1-den-of-evil';
const ranks = { 'sorceress.frost_nova': 1, 'sorceress.ice_bolt': 1 };
const reachXP = [0, 0, 100, 665, 2223, 5423];
const nextXP = [0, 100, 565, 1558, 3200, 5590];
const itemAwards = [5, 6, 6, 6, 7, 8, 9, 10, 10, 11, 11, 12, 12];
const shardAwards = [75, 84, 91, 99, 107, 119, 131, 141, 150, 158, 169, 180, 190];

async function backup(page: Page): Promise<SaveCurrent> {
  await navTo(page, 'settings');
  const pending = page.waitForEvent('download');
  await page.getByTestId('export-save-button').click();
  const download = await pending;
  try { return JSON.parse(await readFile(await download.path(), 'utf8')) as SaveCurrent; }
  finally { await download.delete(); }
}
async function stored(page: Page): Promise<SaveCurrent> {
  return page.evaluate(() => new Promise<SaveCurrent>((resolve, reject) => {
    const open = indexedDB.open('D2H5TextGame');
    open.onerror = () => { reject(new Error('Could not open saved character')); };
    open.onsuccess = () => {
      const db = open.result;
      const get = db.transaction('saves', 'readonly').objectStore('saves').get('current');
      get.onerror = () => { db.close(); reject(new Error('Could not read saved character')); };
      get.onsuccess = () => {
        const record = get.result as { data: SaveCurrent } | undefined;
        db.close();
        if (!record) reject(new Error('No committed character'));
        else resolve(record.data);
      };
    };
  }));
}
async function production(page: Page) {
  expect(await page.evaluate(() => '__GAME__' in window)).toBe(false);
  expect(await page.evaluate(() => navigator.serviceWorker.controller?.state)).toBe('activated');
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)).toBe(true);
}
async function tree(page: Page, skill: string) {
  await navTo(page, 'skills');
  await page.locator('[role="tab"][aria-controls="tabpanel-tree"]').click();
  await page.getByTestId(`skill-node-${skill}`).click();
  return page.getByTestId('skill-detail-panel');
}
async function slots(page: Page, english: boolean, configure = false) {
  await navTo(page, 'skills');
  await page.locator('[role="tab"][aria-controls="tabpanel-active"]').click();
  for (let slot = 1; slot <= 9; slot += 1) {
    const select = page.getByRole('combobox', { name: `${english ? 'Slot' : '槽位'} ${String(slot)}`, exact: true });
    if (configure) await select.selectOption(slot === 1 ? fire : '');
    await expect(select).toHaveValue(slot === 1 ? fire : '');
  }
  await production(page);
}
async function countMonsters(cards: Locator) {
  let count = 0;
  for (const card of await cards.all()) {
    const badge = card.locator(':scope > div:first-of-type > div').filter({ hasText: /^\+\d+$/ });
    expect(await badge.count()).toBeLessThanOrEqual(1);
    count += 1 + (await badge.count() ? Number((await badge.innerText()).slice(1)) : 0);
  }
  return count;
}
async function enter(page: Page, english: boolean, area = 'a1-blood-moor') {
  await navTo(page, 'map');
  await page.getByTestId(`sub-area-row-${area}`)
    .getByRole('button', { name: english ? 'Enter' : '进入', exact: true }).click();
  await expect(page.getByTestId('combat-screen')).toBeVisible();
  expect(await wave(page)).toBe(1);
}
async function wave(page: Page) {
  const value = /^\D*(\d+)/.exec(await page.getByTestId('wave-counter').innerText())?.[1];
  if (!value) throw new Error('Missing numeric wave');
  return Number(value);
}
async function pause(page: Page, english: boolean) {
  await page.getByRole('button', { name: english ? 'Pause' : '暂停', exact: true }).click({ timeout: 500 });
  await expect(page.getByRole('button', { name: english ? 'Resume' : '继续', exact: true }))
    .toHaveAttribute('aria-pressed', 'true');
}
async function flee(page: Page, english: boolean) {
  await page.getByRole('button', { name: english ? 'Flee' : '逃跑', exact: true }).click();
  await expect(page.getByTestId('map-screen')).toBeVisible();
}
async function attach(info: TestInfo, label: string, value: unknown) {
  await info.attach(label, { body: JSON.stringify(value), contentType: 'application/json' });
}
function expectPlayer(actual: SaveCurrent, configured: SaveCurrent, totalXP: number, fireRank: number, spent: number, questPoints = 0) {
  const level = totalXP >= 5423 ? 5 : totalXP >= 2223 ? 4 : totalXP >= 665 ? 3 : totalXP >= 100 ? 2 : 1;
  expect(actual.player).toEqual({ ...configured.player, level, experience: totalXP - (reachXP[level] ?? -1),
    experienceToNextLevel: nextXP[level], statPoints: (level - 1) * 5,
    skillPoints: level - 1 - spent + questPoints,
    skillLevels: { ...ranks, 'sorceress.fire_bolt': fireRank },
    derivedStats: { ...configured.player.derivedStats, lifeMax: 120 + 8 * (level - 1),
      manaMax: 102 + 4 * (level - 1), attack: 22 + 2 * (level - 1), defense: 20 + 3 * (level - 1) } });
}
function expectMerc(actual: SaveCurrent, hired: SaveCurrent, earnedXP: number) {
  const initial = hired.mercs.ownedMercs[0];
  if (!initial) throw new Error('Missing normally hired mercenary');
  let level = 3;
  let xp = earnedXP;
  const threshold = (rank: number) => Math.floor(60 * rank * 1.3 ** (rank - 1));
  while (xp >= threshold(level)) { xp -= threshold(level); level += 1; }
  const gains = level - 3;
  expect(actual.mercs).toEqual({ ...hired.mercs, ownedMercs: [{ ...initial, level,
    coreStats: { ...initial.coreStats, strength: initial.coreStats.strength + gains,
      dexterity: initial.coreStats.dexterity + 3 * gains, energy: initial.coreStats.energy + gains },
    derivedStats: { ...initial.derivedStats, life: initial.derivedStats.life + 8 * gains,
      lifeMax: initial.derivedStats.lifeMax + 8 * gains } }],
    mercProgress: { [merc]: { experience: xp, experienceToNextLevel: threshold(level) } } });
}
async function talk(page: Page, english: boolean, npc: 'akara' | 'charsi', copy?: string, info?: TestInfo) {
  await navTo(page, 'town');
  const trigger = page.getByTestId(`town-npc-${npc}`);
  const box = await trigger.boundingBox();
  expect(box?.width).toBeGreaterThanOrEqual(44);
  expect(box?.height).toBeGreaterThanOrEqual(44);
  await trigger.focus();
  await trigger.press('Enter');
  await expect(trigger).toBeFocused();
  await expect(trigger).toHaveAttribute('aria-expanded', 'true');
  const conversation = page.getByTestId('town-conversation');
  await expect(conversation).toBeVisible();
  if (copy) await expect(conversation.getByRole('status')).toContainText(copy);
  await production(page);
  if (info) await info.attach('den-completed-akara', { body: await page.screenshot({ animations: 'disabled' }), contentType: 'image/png' });
  await page.getByTestId('town-conversation-close').click();
  await expect(conversation).toHaveCount(0);
  await expect(trigger).toBeFocused();
  // Both locales use the same native NPC and focus path.
  await expect(trigger).toHaveAccessibleName(npc === 'akara' ? english ? 'Akara' : '阿卡拉' : english ? 'Charsi' : '查西');
}
async function noRepeat(page: Page, english: boolean, expected: SaveCurrent) {
  await talk(page, english, 'akara', english
    ? 'The darkness in the Den of Evil has lifted. May your new strength guard the camp.'
    : '邪恶巢穴的黑暗已经消散。愿你的新力量守护营地。');
  expect(await backup(page)).toEqual(expected);
  await navTo(page, 'quests');
  await expect(page.getByTestId(`quest-claim-${quest}`)).toBeDisabled();
  await expect(page.getByTestId(`quest-claim-${quest}`)).toHaveText(english ? 'Claimed' : '已领取');
  expect(await backup(page)).toEqual(expected);
}

for (const locale of ['en', 'zh-CN'] as const) {
  test(`ordinary earned Den clear talk claim persists ${locale}`, async ({ page, context }, info) => {
    test.setTimeout(180_000); // Bounded native discovery; no timer acceleration or performance acceptance.
    const english = locale === 'en';
    const errors: string[] = [];
    const watch = (watched: Page) => {
      watched.on('pageerror', error => errors.push(error.message));
      watched.on('console', message => { if (['warning', 'error'].includes(message.type())) errors.push(message.text()); });
    };
    watch(page);
    try {
      await page.clock.setFixedTime(fixedTime);
      await page.goto('/');
      await createCharacter(page, { class: 'sorceress', name });
      await navTo(page, 'settings');
      await page.getByRole('button', { name: english ? 'English' : '简体中文', exact: true }).click();
      await expect.poll(() => page.evaluate(() => navigator.serviceWorker.controller?.state), { timeout: 60_000 }).toBe('activated');
      const fresh = await backup(page);
      await attach(info, 'fresh', fresh);
      expect(fresh.player.skillPoints).toBe(1);
      expect(fresh.player.skillLevels).toEqual(ranks);
      expect(fresh.inventory.backpack).toEqual([]);
      expect(fresh.inventory.equipped).toEqual({});
      const locked = await tree(page, frost);
      await expect(locked.getByRole('button', { name: english ? 'Requires Lv. 6' : '需要等级 6', exact: true })).toBeDisabled();
      expect(await backup(page)).toEqual({ ...fresh, player: { ...fresh.player, comboOrder: [ice] } });
      await navTo(page, 'mercs');
      await page.getByTestId(`hire-merc-${merc}`).getByRole('button', { name: english ? 'Hire' : '招募', exact: true }).click();
      const hired = await backup(page);
      await attach(info, 'hired', hired);
      expect(hired.mercs.fieldedMercId).toBe(merc);
      expect(hired.mercs.ownedMercs).toHaveLength(1);
      expect(hired.mercs.ownedMercs[0]?.level).toBe(3);
      expect(hired.mercs.mercProgress).toEqual({ [merc]: { experience: 0, experienceToNextLevel: 304 } });
      await tree(page, fire);
      await page.getByTestId('skill-detail-panel').getByRole('button', { name: english ? 'Allocate' : '分配', exact: true }).click();
      const learned = await backup(page);
      expect(learned).toEqual({ ...hired, player: { ...hired.player, skillPoints: 0,
        skillLevels: { ...ranks, 'sorceress.fire_bolt': 1 } } });
      await slots(page, english, true);
      const configured = await backup(page);
      await attach(info, 'configured', configured);
      expect(configured).toEqual({ ...learned, player: { ...learned.player, comboOrder: [fire] } });
      let totalXP = 0;
      let items = 0;
      let shards = 0;
      let earned = configured;
      for (let ordinal = 0; ordinal < 13; ordinal += 1) {
        await test.step(`earned prep ordinal ${String(ordinal)}`, async () => {
          await enter(page, english);
          expect(await countMonsters(page.getByTestId('enemies-grid').locator('[data-testid^="unit-card-"]'))).toBe(8 + ordinal);
          await expect(page.getByTestId('allies-grid').locator('[data-kind="merc"]')).toHaveCount(1);
          await page.getByTestId('skip-button').click();
          await expect(page.getByTestId('enemies-grid').locator('[data-testid^="unit-card-"]:not([data-dead="true"])')).toHaveCount(0);
          await expect(page.getByTestId('unit-card-player-1792497600000')).not.toHaveAttribute('data-dead', 'true');
          await expect.poll(() => wave(page), { timeout: 5000, intervals: [50] }).toBe(2);
          await pause(page, english);
          expect(await wave(page)).toBe(2);
          await flee(page, english);
          const before = earned;
          earned = await backup(page);
          // Attach the entire native payload before assertions, including a failed attempt.
          await attach(info, `prep-${String(ordinal)}`, earned);
          totalXP += (8 + ordinal) * 20;
          items += itemAwards[ordinal] ?? 0;
          shards += shardAwards[ordinal] ?? 0;
          expectPlayer(earned, configured, totalXP, 1, 0);
          expectMerc(earned, hired, totalXP * 0.75);
          expect(earned.inventory.backpack).toHaveLength(items);
          expect(earned.inventory.backpack.slice(0, before.inventory.backpack.length)).toEqual(before.inventory.backpack);
          expect(earned.inventory.currencies).toEqual({ 'rune-shard': shards });
          expect(earned.inventory.equipped).toEqual({});
          expect(earned.inventory.stash).toEqual([]);
        });
      }
      expect(totalXP).toBe(3640);
      expect(items).toBe(113);
      expect(shards).toBe(1694);
      expect(earned.player).toMatchObject({ level: 4, experience: 1417, skillPoints: 3, statPoints: 15 });
      expect(earned.mercs.mercProgress[merc]).toEqual({ experience: 1043, experienceToNextLevel: 1336 });
      const detail = await tree(page, fire);
      for (let rank = 2; rank <= 4; rank += 1) {
        await detail.getByRole('button', { name: english ? 'Allocate' : '分配', exact: true }).click();
        await expect(detail.getByText(english ? `Lv. ${String(rank)}/20 · Active` : `等级 ${String(rank)}/20 · 主动`, { exact: true })).toBeVisible();
      }
      const prepared = await backup(page);
      await attach(info, 'prepared', prepared);
      expect(prepared).toEqual({ ...earned, player: { ...earned.player, skillPoints: 0,
        skillLevels: { ...earned.player.skillLevels, 'sorceress.fire_bolt': 4 } } });
      await slots(page, english);
      await navTo(page, 'quests');
      await page.getByTestId(`quest-start-${quest}`).click();
      const tracked = await backup(page);
      await attach(info, 'tracked', tracked);
      const progress = { id: quest, status: 'inProgress', objectives: { 'obj-clear-den': false, 'obj-return': false } } as const;
      expect(tracked).toEqual({ ...prepared, map: { ...prepared.map, questProgress: { ...prepared.map.questProgress, [quest]: progress } } });
      await talk(page, english, 'akara', english
        ? 'Demons remain in the Den of Evil. Slay them all, then return to me.'
        : '邪恶巢穴的恶魔还没有清除。消灭每一只怪物，再回来找我。');
      expect(await backup(page)).toEqual(tracked);
      await enter(page, english, 'a1-den-of-evil');
      const denCounts = [8, 9, 10, 3];
      const denXP = [240, 360, 700, 210];
      const mercXP = [180, 270, 525, 157];
      let mercEarned = 2730;
      let denAward = tracked;
      for (let index = 0; index < 4; index += 1) {
        await test.step(`native Den wave ${String(index + 1)}`, async () => {
          expect(await wave(page)).toBe(index + 1);
          expect(await countMonsters(page.getByTestId('enemies-grid').locator('[data-testid^="unit-card-"]'))).toBe(denCounts[index]);
          await attach(info, `den-opening-${String(index + 1)}`, {
            cards: await page.getByTestId('enemies-grid').innerText(),
            allies: await page.getByTestId('allies-grid').innerText()
          });
          await page.getByTestId('skip-button').click();
          await attach(info, `den-result-${String(index + 1)}`, {
            log: await page.getByTestId('combat-log').innerText(),
            allies: await page.getByTestId('allies-grid').innerText(),
            deadMerc: await page.getByTestId('allies-grid').locator('[data-kind="merc"][data-dead="true"]').count()
          });
          await expect(page.getByTestId('defeat-panel')).toHaveCount(0);
          await expect(page.getByTestId('enemies-grid').locator('[data-testid^="unit-card-"]:not([data-dead="true"])')).toHaveCount(0);
          await expect(page.getByTestId('unit-card-player-1792497600000')).not.toHaveAttribute('data-dead', 'true');
          if (index < 3) {
            await expect.poll(() => wave(page), { timeout: 5000, intervals: [50] }).toBe(index + 2);
            await pause(page, english);
          } else {
            await expect(page.getByTestId('victory-panel')).toBeVisible();
            await page.getByRole('button', { name: english ? 'Return to map' : '返回地图', exact: true }).click();
            await expect(page.getByTestId('cleared-badge-a1-den-of-evil')).toBeVisible();
            await info.attach('den-clear', { body: await page.screenshot({ animations: 'disabled' }), contentType: 'image/png' });
          }
          const before = denAward;
          denAward = await backup(page);
          await attach(info, `den-award-${String(index + 1)}`, denAward);
          totalXP += denXP[index] ?? 0;
          mercEarned += mercXP[index] ?? 0;
          expectPlayer(denAward, configured, totalXP, 4, 3);
          expectMerc(denAward, hired, mercEarned);
          expect(denAward.inventory.backpack.slice(0, before.inventory.backpack.length)).toEqual(before.inventory.backpack);
          expect(denAward.inventory.backpack.length).toBeGreaterThan(before.inventory.backpack.length);
          expect(denAward.inventory.currencies['rune-shard']).toBeGreaterThan(before.inventory.currencies['rune-shard'] ?? -1);
          expect(denAward.inventory.equipped).toEqual({});
          expect(denAward.inventory.stash).toEqual([]);
          if (index < 3) {
            await navTo(page, 'combat');
            expect(await wave(page)).toBe(index + 2);
            await expect(page.getByRole('button', { name: english ? 'Resume' : '继续', exact: true })).toHaveAttribute('aria-pressed', 'true');
          }
        });
      }
      expect(totalXP).toBe(5150);
      expect(mercEarned).toBe(3862);
      expect(denAward.player).toMatchObject({ level: 4, experience: 2927, skillPoints: 0, statPoints: 15 });
      expect(denAward.mercs.mercProgress[merc]).toEqual({ experience: 839, experienceToNextLevel: 2027 });
      expect(denAward.map.questProgress[quest]).toEqual({ ...progress, objectives: { 'obj-clear-den': true, 'obj-return': false } });
      expect(denAward.map.clearedSubAreas).toContain('areas/act1-den-of-evil');
      expect(new Set(denAward.inventory.backpack.map(item => item.id)).size).toBe(denAward.inventory.backpack.length);
      await talk(page, english, 'charsi');
      expect(await backup(page)).toEqual(denAward);
      await talk(page, english, 'akara', english
        ? 'You have cleansed the Den of Evil and given the camp some peace. Claim your reward in Quests.'
        : '你已净化邪恶巢穴，营地重获一线安宁。请到任务中领取应得的奖励。', info);
      const completed = { ...denAward, map: { ...denAward.map, questProgress: { ...denAward.map.questProgress,
        [quest]: { ...progress, status: 'completed', objectives: { 'obj-clear-den': true, 'obj-return': true } } } } } as SaveCurrent;
      const actualCompleted = await backup(page);
      await attach(info, 'actual-den-talk-completed', actualCompleted);
      expect(actualCompleted).toEqual(completed);
      await attach(info, 'den-talk-completed', completed);
      await navTo(page, 'quests');
      const claim = page.getByTestId(`quest-claim-${quest}`);
      await expect(claim).toBeEnabled();
      await claim.focus();
      await claim.press('Enter');
      await expect(claim).toBeDisabled();
      await production(page);
      await info.attach('den-claimed', { body: await page.screenshot({ animations: 'disabled' }), contentType: 'image/png' });
      const claimed = await backup(page);
      await attach(info, 'den-claimed-save', claimed);
      expectPlayer(claimed, configured, 5650, 4, 3, 1);
      expectMerc(claimed, hired, 4237);
      expect(claimed.mercs.mercProgress[merc]).toEqual({ experience: 1214, experienceToNextLevel: 2027 });
      // Known progression above; all other fields must remain byte-for-byte equivalent.
      expect(claimed).toEqual({ ...completed, player: claimed.player, mercs: claimed.mercs,
        map: { ...completed.map, questProgress: { ...completed.map.questProgress,
          [quest]: { ...completed.map.questProgress[quest], rewardClaimed: true } } } });
      await noRepeat(page, english, claimed);
      await page.getByRole('button', { name: english ? 'Save' : '保存', exact: true }).click();
      await expect(page.getByRole('status')).toHaveText(english ? 'Saved' : '已保存');
      const committed = await stored(page);
      await attach(info, 'committed-db', committed);
      expect(committed).toEqual(claimed);
      expect(await backup(page)).toEqual(claimed);
      await page.close();
      const reopened = await context.newPage();
      watch(reopened);
      await reopened.clock.setFixedTime(fixedTime);
      await reopened.goto('/');
      await reopened.getByTestId('home-continue').click();
      await expect(reopened.getByTestId('town-screen')).toBeVisible();
      expect(await stored(reopened)).toEqual(claimed);
      const reopenedSave = await backup(reopened);
      await attach(info, 'actual-reopened', reopenedSave);
      expect(reopenedSave).toEqual(claimed);
      await expect(reopened.getByRole('button', { name: english ? 'English' : '简体中文', exact: true })).toHaveAttribute('aria-pressed', 'true');
      await noRepeat(reopened, english, claimed);
      await slots(reopened, english);
      await production(reopened);
      expect(errors).toEqual([]);
    } finally { await attach(info, 'browser-errors', errors); }
  });
}
