async function backup(page: Page) {
  await navTo(page, 'settings');
  const pending = page.waitForEvent('download');
  await page.getByTestId('export-save-button').click();
  const download = await pending;
  try { return JSON.parse(await readFile(await download.path(), 'utf8')); }
  finally { await download.delete(); }
}
async function disk(page: Page) {
  return page.evaluate(async () => {
    const db = await new Promise<IDBDatabase>((resolve, reject) => {
      const req = indexedDB.open('D2H5TextGame');
      req.onsuccess = () => resolve(req.result); req.onerror = () => reject(req.error);
    });
    try {
      const record = await new Promise<{ data: unknown }>((resolve, reject) => {
        const req = db.transaction('saves', 'readonly').objectStore('saves').get('current');
        req.onsuccess = () => resolve(req.result); req.onerror = () => reject(req.error);
      });
      return JSON.parse(JSON.stringify(record.data, (_key, value) => value instanceof Map ? [...value.entries()] : value));
    } finally { db.close(); }
  });
}
async function roster(page: Page, locale: string) {
  await navTo(page, 'mercs');
  const card = page.getByTestId(`merc-card-${mercId}`);
  for (const [label, value] of [['LVL', 16], ['STR', 25], ['DEX', 61], ['VIT', 16]]) {
    const stat = card.getByText(String(label), { exact: true }).locator('..');
    await stat.scrollIntoViewIfNeeded(); await expect(stat).toHaveText(`${label}${value}`);
  }
  await expect(card.getByTestId(`merc-xp-${mercId}`)).toBeVisible();
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)).toBe(true);
  expect(await page.evaluate(() => '__GAME__' in window)).toBe(false);
  await card.getByText('DEX', { exact: true }).scrollIntoViewIfNeeded();
  await page.screenshot({ path: `${root}/merc-${locale}-${page.viewportSize()?.width}.png` });
}
