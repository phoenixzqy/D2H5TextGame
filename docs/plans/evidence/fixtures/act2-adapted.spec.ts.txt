  }

 }catch(error){checks.error=error instanceof Error?error.stack:String(error);checks.phase=phase;checks.entered=entered;if(!active.isClosed())try{const p=active.getByRole('button',{name:'Pause',exact:true});if(await p.count())await p.click({timeout:1000});await active.screenshot({path:root+'/last-page.png'});await active.keyboard.press('Escape');await save('last-save',await backup(active));}catch(e){checks.diagnosticError=String(e);}throw error;
 }finally{checks.pageErrors=errors;await writeFile(root+'/assertions.json',JSON.stringify(checks,null,2));if(!active.isClosed())await active.close();}
});
