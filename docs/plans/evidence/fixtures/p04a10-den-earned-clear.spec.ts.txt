async function backup(page: Page): Promise<SaveCurrent> {
  await navTo(page, 'settings');
  const pending = page.waitForEvent('download');
  await page.getByTestId('export-save-button').click();
  const download = await pending;
  try { return JSON.parse(await readFile(await download.path(), 'utf8')) as SaveCurrent; }
  finally { await download.delete(); }
}
async function stored(page: Page): Promise<SaveCurrent> {
  return page.evaluate(() => new Promise<SaveCurrent>((resolve, reject) => {
    const open = indexedDB.open('D2H5TextGame');
    open.onerror = () => { reject(new Error('Could not open saved character')); };
    open.onsuccess = () => {
      const db = open.result;
      const get = db.transaction('saves', 'readonly').objectStore('saves').get('current');
      get.onerror = () => { db.close(); reject(new Error('Could not read saved character')); };
      get.onsuccess = () => {
        const record = get.result as { data: SaveCurrent } | undefined;
        db.close();
        if (!record) reject(new Error('No committed character'));
        else resolve(record.data);
      };
    };
  }));
}
async function production(page: Page) {
  expect(await page.evaluate(() => '__GAME__' in window)).toBe(false);
  expect(await page.evaluate(() => navigator.serviceWorker.controller?.state)).toBe('activated');
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)).toBe(true);
}
async function tree(page: Page, skill: string) {
  await navTo(page, 'skills');
  await page.locator('[role="tab"][aria-controls="tabpanel-tree"]').click();
  await page.getByTestId(`skill-node-${skill}`).click();
  return page.getByTestId('skill-detail-panel');
}
async function slots(page: Page, english: boolean, selected: readonly string[]) {
  await navTo(page, 'skills');
  await page.locator('[role="tab"][aria-controls="tabpanel-active"]').click();
  for (let slot = 1; slot <= 9; slot += 1) {
    const select = page.getByRole('combobox', { name: `${english ? 'Slot' : '槽位'} ${String(slot)}`, exact: true });
    await select.selectOption(selected[slot - 1] ?? '');
    await expect(select).toHaveValue(selected[slot - 1] ?? '');
  }
  await production(page);
}
async function countMonsters(cards: Locator) {
  let count = 0;
  for (const card of await cards.all()) {
    const badge = card.locator(':scope > div:first-of-type > div').filter({ hasText: /^\+\d+$/ });
    expect(await badge.count()).toBeLessThanOrEqual(1);
    count += 1 + (await badge.count() ? Number((await badge.innerText()).slice(1)) : 0);
  }
  return count;
}
async function enter(page: Page, english: boolean, area = 'a1-blood-moor') {

async function wave(page: Page) {
  const value = /^\D*(\d+)/.exec(await page.getByTestId('wave-counter').innerText())?.[1];
  if (!value) throw new Error('Missing numeric wave');
  return Number(value);
}

async function attach(info: TestInfo, label: string, value: unknown) {

/** One readonly DOM observation before native inter-wave navigation can replace it. */
async function terminalDOM(page: Page) {
  return page.evaluate(() => {
    const hero = document.querySelector('[data-testid="unit-card-player-1792497600000"]');
    const log = Array.from(document.querySelectorAll('[data-testid="combat-log"] > div > span:last-child'), node => node.textContent);
    return {
      waveText: document.querySelector('[data-testid="wave-counter"]')?.textContent,
      waveKind: document.querySelector('[data-testid="wave-kind-label"]')?.textContent,
      heroPresent: hero !== null,
      heroDead: hero?.getAttribute('data-dead') ?? null,
      liveEnemies: document.querySelectorAll('[data-testid="enemies-grid"] [data-testid^="unit-card-"]:not([data-dead="true"])').length,
      defeatPanels: document.querySelectorAll('[data-testid="defeat-panel"]').length,
      deadMerc: document.querySelectorAll('[data-testid="allies-grid"] [data-kind="merc"][data-dead="true"]').length,
      skipDisabled: document.querySelector<HTMLButtonElement>('[data-testid="skip-button"]')?.disabled,
      log,
      mp: document.querySelector('[role="group"][aria-label="MP"] [role="progressbar"]')?.getAttribute('aria-valuenow'),
      cards: Array.from(document.querySelectorAll('[data-testid^="unit-card-"]'), card => ({
        id: card.getAttribute('data-testid'), kind: card.getAttribute('data-kind'),
        dead: card.getAttribute('data-dead'), hp: card.querySelector('.bg-red-600')?.getAttribute('style'), text: card.textContent
      })),
      allies: document.querySelector('[data-testid="allies-grid"]')?.textContent,
      enemies: document.querySelector('[data-testid="enemies-grid"]')?.textContent
    };
  });
}
function expectTerminal(result: Awaited<ReturnType<typeof terminalDOM>>, expectedWave: number) {
  expect(Number(/^\D*(\d+)/.exec(result.waveText ?? '')?.[1])).toBe(expectedWave);
  expect(result.skipDisabled).toBe(true);
  expect(result.log.at(-1)).toBe('Victory! All enemies are defeated.');
  expect(result.defeatPanels).toBe(0);
  expect(result.liveEnemies).toBe(0);
  expect(result.heroPresent).toBe(true);
  expect(result.heroDead).not.toBe('true');
}
function expectPlayer(actual: SaveCurrent, configured: SaveCurrent, totalXP: number, fireRank: number, spent: number, questPoints = 0) {
  const level = totalXP >= 11013 ? 6 : totalXP >= 5423 ? 5 : totalXP >= 2223 ? 4 : totalXP >= 665 ? 3 : totalXP >= 100 ? 2 : 1;
  expect(actual.player).toEqual({ ...configured.player, level, experience: totalXP - (reachXP[level] ?? -1),
    experienceToNextLevel: nextXP[level], statPoints: (level - 1) * 5,
    skillPoints: level - 1 - spent + questPoints,
    skillLevels: { ...ranks, 'sorceress.fire_bolt': fireRank },
    derivedStats: { ...configured.player.derivedStats, lifeMax: 120 + 8 * (level - 1),
      manaMax: 102 + 4 * (level - 1), attack: 22 + 2 * (level - 1), defense: 20 + 3 * (level - 1) } });
}
function expectMerc(actual: SaveCurrent, hired: SaveCurrent, earnedXP: number, equipment = hired.mercs.mercEquipment) {
  const initial = hired.mercs.ownedMercs[0];
  if (!initial) throw new Error('Missing normally hired mercenary');
  let level = 3;
  let xp = earnedXP;
  const threshold = (rank: number) => Math.floor(60 * rank * 1.3 ** (rank - 1));
  while (xp >= threshold(level)) { xp -= threshold(level); level += 1; }
  const gains = level - 3;
  expect(actual.mercs).toEqual({ ...hired.mercs, mercEquipment: equipment, ownedMercs: [{ ...initial, level,
    coreStats: { ...initial.coreStats, strength: initial.coreStats.strength + gains,
      dexterity: initial.coreStats.dexterity + 3 * gains, energy: initial.coreStats.energy + gains },
    derivedStats: { ...initial.derivedStats, life: initial.derivedStats.life + 8 * gains,
      lifeMax: initial.derivedStats.lifeMax + 8 * gains } }],
    mercProgress: { [merc]: { experience: xp, experienceToNextLevel: threshold(level) } } });
}

        await attach(info, `den-terminal-${String(index + 1)}`, terminal);
        expectTerminal(terminal, index + 1); // Stop at first nonwin, no retry/build change.
        previousTerminal = terminal;
        if (index < 3) {
          await expect.poll(() => wave(page), { timeout: 5000, intervals: [50] }).toBe(index + 2);
          await pause(page, english);
        } else {
          await expect(page.getByTestId('victory-panel')).toBeVisible();
          await page.getByTestId('return-to-map').click();
          await expect(page.getByTestId('cleared-badge-a1-den-of-evil')).toBeVisible();
          await expect(page.getByTestId('farm-here-a1-den-of-evil')).toBeEnabled();
          await info.attach('actual-earned-den-clear', { body: await page.screenshot(), contentType: 'image/png' });
        }
        const before = latest;
        latest = await backup(page);
        await attach(info, `den-award-${String(index + 1)}`, latest);
        totalXP += awards[index] ?? 0;
        mercEarned += mercAwards[index] ?? 0;
        expect(latest.player).toEqual({ ...geared.player, experience: totalXP - 11013 });
        expectMerc(latest, hired, mercEarned, upgradedMercEquipment);
        expect(latest.inventory.backpack.slice(0, before.inventory.backpack.length)).toEqual(before.inventory.backpack);
        expect(latest.inventory.backpack.length).toBeGreaterThan(before.inventory.backpack.length);
        expect(new Set(latest.inventory.backpack.map(item => item.id)).size).toBe(latest.inventory.backpack.length);
        expect(latest.inventory.currencies['rune-shard']).toBeGreaterThan(before.inventory.currencies['rune-shard'] ?? 0);
        expect(latest.inventory.equipped).toEqual(upgradedEquipment);
        expect(latest.inventory.stash).toEqual([]);
        expect(latest.map.questProgress).toEqual({ ...tracked.map.questProgress, [quest]: {
          id: quest, status: 'inProgress', objectives: { 'obj-clear-den': index === 3, 'obj-return': false }
        } });
        expect(latest.map.clearedSubAreas).toEqual(index === 3 ? ['areas/act1-den-of-evil'] : []);
        if (index < 3) {
          await navTo(page, 'combat');
          expect(await wave(page)).toBe(index + 2);
          await expect(page.getByRole('button', { name: 'Resume', exact: true })).toHaveAttribute('aria-pressed', 'true');
        }
      });
    }
    expect(totalXP).toBe(12730);
    expect(mercEarned).toBe(9546);
    expect(latest.player).toMatchObject({ level: 6, experience: 1717, statPoints: 0, skillPoints: 0 });
    expect(latest.mercs.mercProgress[merc]).toEqual({ experience: 1485, experienceToNextLevel: 4404 });
    await navTo(page, 'quests');
    await expect(page.getByTestId(`quest-claim-${quest}`)).toHaveCount(0);
    await navTo(page, 'town');
    await page.getByTestId('town-npc-akara').click();
    await expect(page.getByRole('status')).toContainText('Claim your reward in Quests.');
    const talked = await backup(page);
    expect(talked).toEqual({ ...latest, map: { ...latest.map, questProgress: {
      ...latest.map.questProgress, [quest]: { id: quest, status: 'completed',
        objectives: { 'obj-clear-den': true, 'obj-return': true } }
    } } });
    await attach(info, 'den-talk-no-award', talked);
    await navTo(page, 'quests');
    const claim = page.getByTestId(`quest-claim-${quest}`);
    await expect(claim).toBeEnabled();
    await claim.click();
    await expect(claim).toBeDisabled();
    const claimed = await backup(page);
    expect(claimed).toEqual({ ...talked, player: { ...talked.player, experience: 2217, skillPoints: 1 },
      map: { ...talked.map, questProgress: { ...talked.map.questProgress, [quest]: {
        id: quest, status: 'completed', objectives: { 'obj-clear-den': true, 'obj-return': true }, rewardClaimed: true
      } } }
    });
    await attach(info, 'den-once-only-earned-award', claimed);
    await navTo(page, 'quests');
    await expect(claim).toBeDisabled();
    await production(page);
    await info.attach('actual-mobile-den-claim', { body: await page.screenshot(), contentType: 'image/png' });
    await navTo(page, 'town');
    await page.getByTestId('town-npc-akara').click();
    await page.getByTestId('town-conversation-quests').click();
    await expect(claim).toBeDisabled();
    expect(await backup(page)).toEqual(claimed);
    await page.getByRole('button', { name: 'Save', exact: true }).click();
    await expect(page.getByRole('status')).toHaveText('Saved');
    const committed = await stored(page);
    await attach(info, 'actual-den-committed', committed);
    expect(committed).toEqual(claimed);
    expect(await backup(page)).toEqual(committed);
    await page.close();
    const reopened = await context.newPage(); active = reopened;
    watch(reopened);
    await reopened.setViewportSize({ width: 360, height: 640 });
    expect(reopened.viewportSize()).toEqual({ width: 360, height: 640 });
    await reopened.clock.setFixedTime(fixedTime);
    await reopened.goto('/');
    await reopened.getByTestId('home-continue').click();
    await expect(reopened.getByTestId('town-screen')).toBeVisible();
    expect(await stored(reopened)).toEqual(committed);
    const continued = await backup(reopened);
    await attach(info, 'actual-den-new-page-continue', continued);
    expect(continued).toEqual(committed);
    await navTo(reopened, 'map');
    await expect(reopened.getByTestId('cleared-badge-a1-den-of-evil')).toBeVisible();
    await expect(reopened.getByTestId('farm-here-a1-den-of-evil')).toBeEnabled();
    await navTo(reopened, 'quests');
    await expect(reopened.getByTestId(`quest-claim-${quest}`)).toBeDisabled();
    await production(reopened);
    await info.attach('actual-mobile-den-continued', { body: await reopened.screenshot(), contentType: 'image/png' });
    await reopened.setViewportSize({ width: 1280, height: 800 });
    expect(reopened.viewportSize()).toEqual({ width: 1280, height: 800 });
    await navTo(reopened, 'map');
    await expect(reopened.getByTestId('cleared-badge-a1-den-of-evil')).toBeVisible();
    await production(reopened);
    expect(await backup(reopened)).toEqual(committed);
    await info.attach('actual-desktop-den-continued', { body: await reopened.screenshot(), contentType: 'image/png' });
    expect(errors).toEqual([]);
  } finally {
    for (const capture of [
      async () => { await attach(info, 'browser-errors', errors); },
      async () => { if (!active.isClosed()) await attach(info, 'last-native-DOM', await terminalDOM(active)); },
      async () => { if (!active.isClosed()) await info.attach('last-native-screen', { body: await active.screenshot(), contentType: 'image/png' }); }
    ]) {
      try { await capture(); }
      catch (error) {
        diagnosticFailure ??= error instanceof Error ? error : new Error(String(error));
        console.error('Den diagnostic capture failed:', error);
      }
    }
  }
  if (diagnosticFailure !== undefined) throw diagnosticFailure;
});
