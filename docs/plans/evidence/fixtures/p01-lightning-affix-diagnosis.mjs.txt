function account(wave){
 const units=new Map([...wave.opening.playerTeam,...wave.opening.enemyTeam].map(u=>[u.id,{id:u.id,side:u.side,opening:{hp:u.life,mp:u.mana},hp:u.life,mp:u.mana,hpMax:u.stats.lifeMax,mpMax:u.stats.manaMax,damage:{},damageEffective:{},overkill:0,healEffective:0,hpOrbEffective:0,hpOrbCount:0,mpOrbCount:0,manaDelta:{},actions:[],resources:[]}]));
 let action=null;
 for(const [index,event] of wave.opening.recordedEvents.entries()){
  if(event.kind==='action'){ action={actor:event.actor,skillId:event.skillId,simClockMs:event.simClockMs}; units.get(event.actor)?.actions.push(action); }
  const u=units.get(event.target); if(!u) continue;
  const before={hp:u.hp,mp:u.mp};
  if((event.kind==='damage'&&!event.dodged)||event.kind==='dot'){
   const amount=event.amount; const effective=Math.min(u.hp,amount);
   const key=event.kind==='dot'?'dot':`${event.source}:${event.damageType}`;
   u.damage[key]=(u.damage[key]??0)+amount; u.damageEffective[key]=(u.damageEffective[key]??0)+effective;
   u.overkill+=amount-effective;u.hp=Math.max(0,u.hp-amount);
  }else if(event.kind==='heal'){
   const effective=Math.min(u.hpMax-u.hp,event.amount);u.healEffective+=effective;u.hp+=effective;
  }else if(event.kind==='orb'){
   if(event.orb.kind==='hp'){const effective=Math.min(u.hpMax-u.hp,Math.floor(u.hpMax*event.orb.restorePct));u.hpOrbEffective+=effective;u.hpOrbCount++;u.hp+=effective;}
   else u.mpOrbCount++;
  }else if(event.kind==='mana'){
   const delta=event.mana-u.mp;u.manaDelta[event.reason]=(u.manaDelta[event.reason]??0)+delta;u.mp=event.mana;
  }else if(event.kind==='death'){
   if(u.hp!==0) throw new Error(`Death without accounted damage: ${u.id}, HP ${u.hp}`);
  }
  if(before.hp!==u.hp||before.mp!==u.mp||['damage','dot','orb','heal','death'].includes(event.kind)) u.resources.push({index,clock:event.simClockMs,event,contextAction:action,before,after:{hp:u.hp,mp:u.mp}});
 }
 for(const final of [...wave.opening.outcome.finalPlayerTeam,...wave.opening.outcome.finalEnemyTeam]){
  const u=units.get(final.id);assert(u,`Unregistered final unit ${final.id}`);u.actualFinal={hp:final.life,mp:final.mana};
  assert.equal(u.hp,final.life,`HP accounting ${final.id}`);assert.equal(u.mp,final.mana,`MP accounting ${final.id}`);
  u.reconciled=true;
 }
 return [...units.values()];
}
