const beforeSources = await sources();
const report = { kind: 'new paid Ice Blast route screen from exact service-earned L12; not native UI proof',
  head: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: wt, encoding: 'utf8' }).trim(),
  inputSha: sha(input), scriptSha: sha(await readFile(fileURLToPath(import.meta.url))),
  node: process.version, pid: process.pid, seed: 2223645302, firstOrdinal: 102, maxFarmRuns: 0,

    'earned points only: Ice Blast; each level 2 Vitality / 3 Energy',

  runs: [], allocations: [], totalCombatMs: 0, sources: beforeSources };
let server, restorers = [], state, helpers, player, inventory, merc, map, meta, formation, hydration, combat, oldPity;
const originalNow = Date.now;
try {
  await mkdir(path.join(root, 'tmp'), { recursive: true });
  server = await createServer({ configFile: false, root: wt, cacheDir: path.join(root, 'vite-cache'),
    server: { middlewareMode: true, hmr: false }, resolve: { alias: { '@': path.join(wt, 'src') } } });
  const load = rel => server.ssrLoadModule('/' + rel);
  state = await load('tests/sim/production/earned-state.ts');
  const playback = await load('tests/sim/production/earned-playback.ts');
  helpers = await load('src/stores/combatHelpers.ts');
  const pity = await load('src/stores/idleElitePity.ts');
  const { decodeSave } = await load('src/engine/save/validate.ts');
  const { toJsonSafe } = await load('src/engine/save/serialize.ts');
  const { xpTotal } = await load('src/engine/progression/xp.ts');
  ({ usePlayerStore: player } = await load('src/stores/playerStore.ts'));
  ({ useInventoryStore: inventory } = await load('src/stores/inventoryStore.ts'));
  ({ useMercStore: merc } = await load('src/stores/mercStore.ts'));
  ({ useMapStore: map } = await load('src/stores/mapStore.ts'));

  ({ useFormationStore: formation } = await load('src/stores/formationStore.ts'));
  ({ useHydrationStore: hydration } = await load('src/stores/hydrationStore.ts'));
  ({ useCombatStore: combat } = await load('src/stores/combatStore.ts'));
